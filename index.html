'topSeller' : 'topAchiever' ); const sound = document.getElementById('celebrateSound'); // Counter & progress cnt.textContent = `${total} m²`; const pct = Math.min((total / target) * 100, 100); prog.style.width = `${pct}%`; lbl.textContent = `Target: ${target} m²`; // Milestones (تقدر تضبط النقاط كما تريد) if (total >= target) { mile.textContent = '🎉 Goal Achieved!'; sound.play(); } else if (total >= target * 0.75) { mile.textContent = '🔥 Almost there!'; } else if (total >= target * 0.5) { mile.textContent = '💪 Halfway!'; } else { mile.textContent = `Let's break the 1st ${Math.floor(target*0.25)} m²!`; } // Top performer const top = Object.entries(byMember) .sort((a,b) => b[1] - a[1])[0]; if (top) { topEl.textContent = isOps ? `Top Achiever: ${top[0]}` : `Top Seller: ${top[0]}`; } } fetchData(); setInterval(fetchData, 60_000);
